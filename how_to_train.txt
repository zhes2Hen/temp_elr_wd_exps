应该如何8卡分配给4个实验训练，下面是参考启动方式：

schedules=(cos wsd10 wsd30 7111)
pairs=(0,1 2,3 4,5 6,7)
pids=()

mkdir -p ../out/nlq_sweep/launcher_logs
for i in "${!schedules[@]}"; do
    lrs="${schedules[$i]}"
    CUDA_VISIBLE_DEVICES="${pairs[$i]}" \
        torchrun --standalone --nproc-per-node=2 train.py \
        "config/nlq_sweep/train_nlq_adamw_lr1p2e-3_wd0.1_${lrs}.py" \
        > "../out/nlq_sweep/launcher_logs/${lrs}.log" 2>&1 &
    pids+=("$!")
done

failed=0
for pid in "${pids[@]}"; do wait "$pid" || failed=1; done
test "$failed" -eq 0
